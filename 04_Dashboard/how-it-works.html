import { range } from "./js/govuk-inputs.js"
import { dots } from "./js/dots.js"
meta = FileAttachment("data/meta.json").json()
scores = FileAttachment("data/scores.json").json()
total = (a) => a.reduce((x, y) => x + y, 0)
// Confusion matrix of the held-out set at threshold t: bins at or above t are flagged serious.
confusion = (t) => {
  const k = Math.round(t * scores.bins);
  const tp = total(scores.serious.slice(k)), fn = total(scores.serious.slice(0, k));
  const fp = total(scores.slight.slice(k)), tn = total(scores.slight.slice(0, k));
  const n = tp + fn + fp + tn;
  return {tp, fn, fp, tn, n, recall: tp / (tp + fn),
          precision: tp + fp ? tp / (tp + fp) : 0, flagged: (tp + fp) / n};
}
fmt = (x) => x.toLocaleString("en-GB")
per100 = (x) => Math.round(x * 100)
base = confusion(meta.threshold)
html`<span class="govuk-caption-m">Road collision triage</span>
<h1 class="govuk-heading-l">How the estimate works</h1>
<p class="govuk-body-l">At the threshold of ${meta.threshold}, the tool catches
<strong>${per100(base.recall)} of every 100</strong> serious collisions. The price is that
${100 - per100(base.precision)} of every 100 Category 2 ambulances go to a collision that turns
out to be slight.</p>`

Where to draw the line

The model gives every call a probability. Above the threshold the call is treated as serious. Slide left and no serious collision is missed, because every call gets a Category 2 ambulance. Slide right and the ambulances stay free for the serious collisions the model no longer spots.
